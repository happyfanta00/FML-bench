"""
Improved DP training algorithm for CIFAR-10 with Opacus DP-SGD.

Changes vs. baseline:
- Architecture follows the end-to-end DP CNN of Tramer & Boneh (2021):
  conv(32)x2 -> pool -> conv(64)x2 -> pool -> conv(128)x2 -> pool -> fc(128) -> fc(10).
- Tanh activations ("tempered sigmoid", Papernot et al. 2020). Bounded
  activations keep per-sample gradient norms small, so less signal is lost
  to clipping.
- GroupNorm for training stability (BatchNorm is not DP-compatible).
- In-model random horizontal flip during training. Each flip is applied to a
  single sample, so it does not affect the privacy accounting.
- Smaller clipping norm (0.1) with a larger learning rate. This is the
  standard recipe for DP-SGD, where nearly every per-sample gradient is
  clipped.
- Slightly more epochs, which the privacy accountant compensates for with
  more noise.
"""
import torch
import torch.nn as nn


def _conv_block(cin, cout, groups=16):
    return [
        nn.Conv2d(cin, cout, 3, padding=1),
        nn.GroupNorm(min(groups, cout), cout),
        nn.Tanh(),
    ]


class DPCNN(nn.Module):
    """Tramer & Boneh style CNN with GroupNorm and tanh activations."""

    def __init__(self, num_classes=10):
        super().__init__()
        layers = []
        layers += _conv_block(3, 32)
        layers += _conv_block(32, 32)
        layers += [nn.MaxPool2d(2)]
        layers += _conv_block(32, 64)
        layers += _conv_block(64, 64)
        layers += [nn.MaxPool2d(2)]
        layers += _conv_block(64, 128)
        layers += _conv_block(128, 128)
        layers += [nn.AdaptiveAvgPool2d((4, 4))]
        self.features = nn.Sequential(*layers)
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 4 * 4, 128),
            nn.Tanh(),
            nn.Linear(128, num_classes),
        )

    @staticmethod
    def _random_flip(x):
        # Per-sample random horizontal flip (only applied in training mode).
        if x.dim() != 4:
            return x
        mask = (torch.rand(x.shape[0], 1, 1, 1, device=x.device) < 0.5)
        return torch.where(mask, torch.flip(x, dims=[3]), x)

    def forward(self, x):
        if self.training:
            x = self._random_flip(x)
        x = self.features(x)
        x = self.classifier(x)
        return x


def get_model(num_classes=10):
    """Return the model to be trained with DP-SGD."""
    return DPCNN(num_classes=num_classes)


def get_training_config():
    """Return training hyperparameters."""
    return {
        'epochs': 30,
        'lr': 2.0,
        'momentum': 0.9,
        'max_grad_norm': 0.1,
        'target_epsilon': 8.0,
        'target_delta': 1e-5,
        'batch_size': 512,
    }

- Larger batch size (1024). The noise on the averaged gradient scales as

  sigma, which gives a better signal-to-noise ratio per step.

        nn.GroupNorm(min(groups, cout), cout),
