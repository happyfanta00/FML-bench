"""
Improved semi-supervised learning algorithm: FixMatch on CIFAR-100.

Changes vs. the baseline:
1. SSL-tuned WideResNet (USB/FixMatch style): LeakyReLU(0.1), lower BN momentum
   and kaiming/xavier initialisation. This gives more stable BN statistics when
   labeled and unlabeled batches are mixed.
2. A full RandAugment (14 PIL ops, random magnitudes) plus Cutout as the
   strong augmentation. This is the augmentation used in the original FixMatch
   paper. The baseline's 6-op version was far weaker.
3. A lower confidence threshold (0.7). With only 4 labels per class on
   CIFAR-100, predictions rarely exceed 0.95, so almost no unlabeled data
   contributed under the baseline setting.
"""
import random

import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.transforms as transforms
import torchvision.transforms.functional as TF
import numpy as np
from PIL import Image, ImageOps, ImageEnhance, ImageDraw


BN_MOMENTUM = 0.01
LEAKY_SLOPE = 0.1


class WideResNetBlock(nn.Module):
    """Basic block for WideResNet."""

    def __init__(self, in_channels, out_channels, stride=1, dropout_rate=0.0):
        super().__init__()
        self.bn1 = nn.BatchNorm2d(in_channels, momentum=BN_MOMENTUM)
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3, stride=stride, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels, momentum=BN_MOMENTUM)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3, padding=1, bias=False)
        self.dropout = nn.Dropout(dropout_rate) if dropout_rate > 0 else nn.Identity()

        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Conv2d(in_channels, out_channels, 1, stride=stride, bias=False)

    def forward(self, x):
        out = F.leaky_relu(self.bn1(x), LEAKY_SLOPE)
        shortcut = self.shortcut(out)
        out = self.conv1(out)
        out = self.dropout(F.leaky_relu(self.bn2(out), LEAKY_SLOPE))
        out = self.conv2(out)
        return out + shortcut


class WideResNet(nn.Module):
    """WideResNet-28-2 for CIFAR."""

    def __init__(self, depth=28, widen_factor=2, num_classes=100, dropout_rate=0.0):
        super().__init__()
        assert (depth - 4) % 6 == 0
        n = (depth - 4) // 6
        channels = [16, 16 * widen_factor, 32 * widen_factor, 64 * widen_factor]

        self.conv1 = nn.Conv2d(3, channels[0], 3, padding=1, bias=False)
        self.layer1 = self._make_layer(n, channels[0], channels[1], 1, dropout_rate)
        self.layer2 = self._make_layer(n, channels[1], channels[2], 2, dropout_rate)
        self.layer3 = self._make_layer(n, channels[2], channels[3], 2, dropout_rate)
        self.bn = nn.BatchNorm2d(channels[3], momentum=BN_MOMENTUM)
        self.fc = nn.Linear(channels[3], num_classes)

        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='leaky_relu')
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.constant_(m.weight, 1.0)
                nn.init.constant_(m.bias, 0.0)
            elif isinstance(m, nn.Linear):
                nn.init.xavier_normal_(m.weight)
                nn.init.constant_(m.bias, 0.0)

    def _make_layer(self, n, in_channels, out_channels, stride, dropout_rate):
        layers = [WideResNetBlock(in_channels, out_channels, stride, dropout_rate)]
        for _ in range(1, n):
            layers.append(WideResNetBlock(out_channels, out_channels, 1, dropout_rate))
        return nn.Sequential(*layers)

    def forward(self, x):
        out = self.conv1(x)
        out = self.layer1(out)
        out = self.layer2(out)
        out = self.layer3(out)
        out = F.leaky_relu(self.bn(out), LEAKY_SLOPE)
        out = F.adaptive_avg_pool2d(out, 1).flatten(1)
        return self.fc(out)


def get_model(num_classes=100):

def _cutout(img, frac):
    """Gray-fill a random square whose side is frac * image width."""
    if frac <= 0:
        return img
    w, h = img.size
    size = int(frac * w)
    if size <= 0:
        return img
    x0 = int(np.random.uniform(0, w))
    y0 = int(np.random.uniform(0, h))
    x0 = int(max(0, x0 - size / 2.0))
    y0 = int(max(0, y0 - size / 2.0))
    x1 = min(w, x0 + size)
    y1 = min(h, y0 + size)
    img = img.copy()
    if img.mode == 'L':
        fill = 127
    else:
        fill = (127, 127, 127)[:len(img.getbands())] if len(img.getbands()) <= 3 else (127, 127, 127, 255)
    ImageDraw.Draw(img).rectangle([x0, y0, x1, y1], fill=fill)
    return img


class RandAugment:
    """RandAugment (FixMatch-style) with random magnitudes plus Cutout."""

    def __init__(self, n=2, m=10):
        self.n = n
        self.m = m
        self.ops = [
            _identity, _autocontrast, _equalize, _brightness, _color,
            _contrast, _sharpness, _posterize, _solarize, _rotate,
            _shear_x, _shear_y, _translate_x, _translate_y,
        ]
        # Kept for backward compatibility with any code inspecting this attribute.
        self.augmentations = self.ops

    def _apply_pil(self, img):
        if img.mode not in ('RGB', 'L'):
            img = img.convert('RGB')
        ops = random.sample(self.ops, self.n)
        for op in ops:
            v = np.random.uniform(0.05, 1.0)
            try:
                img = op(img, v)
            except Exception:
                pass
        img = _cutout(img, np.random.uniform(0.0, 0.5))
        return img

    def __call__(self, img):
        if isinstance(img, Image.Image):
            return self._apply_pil(img)
        if isinstance(img, torch.Tensor):
            is_uint8 = img.dtype == torch.uint8
            pil = TF.to_pil_image(img)
            out = self._apply_pil(pil)
            if is_uint8:
                return TF.pil_to_tensor(out)
            return TF.to_tensor(out).to(img.dtype)
        if isinstance(img, np.ndarray):
            pil = Image.fromarray(img.astype(np.uint8))
            return np.asarray(self._apply_pil(pil))
        return img
