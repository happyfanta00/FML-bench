"""
OOD detection algorithm: LogitNorm training + MSP scoring.

LogitNorm (Wei et al., 2022) normalizes the logit vector to a constant norm
(1/tau) during training. Plain cross-entropy keeps increasing the logit norm
to reduce the loss, which yields overconfident predictions even on OOD inputs.
Decoupling the norm from the training objective gives much better separation
between ID and OOD confidence, especially for far-OOD data (e.g. SVHN).

Implementation: the model's forward() returns normalized logits / tau while in
training mode, so the framework's standard cross-entropy becomes the LogitNorm
loss. In eval mode, forward() returns raw logits (argmax is unchanged, so
accuracy is unaffected), and the OOD score is MSP on the raw logits.
"""
import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.models as models


LOGITNORM_TAU = 0.04


class ResNet18CIFAR(nn.Module):
    """ResNet-18 adapted for CIFAR-10 (32x32 images) with LogitNorm training."""

    def __init__(self, num_classes=10, tau=LOGITNORM_TAU):
        super().__init__()
        backbone = models.resnet18(weights=None)
        # Adapt for CIFAR: smaller conv1, no maxpool
        backbone.conv1 = nn.Conv2d(3, 64, 3, 1, 1, bias=False)
        backbone.maxpool = nn.Identity()
        self.feature_dim = backbone.fc.in_features
        backbone.fc = nn.Identity()
        self.backbone = backbone
        self.fc = nn.Linear(self.feature_dim, num_classes)
        self.tau = tau

    def forward_raw(self, x):
        features = self.backbone(x)
        return self.fc(features)

    def forward(self, x):
        logits = self.forward_raw(x)
        if self.training:
            # LogitNorm: CE on logits normalized to norm 1/tau
            norms = torch.norm(logits, p=2, dim=-1, keepdim=True) + 1e-7
            logits = logits / (norms * self.tau)
        return logits

    def get_features(self, x):
        """Extract penultimate features."""
        return self.backbone(x)


def get_model(num_classes=10):
    """Return classifier model."""
    return ResNet18CIFAR(num_classes=num_classes)


def get_training_config():
    """Return training hyperparameters."""
    return {
        'epochs': 100,
        'batch_size': 128,
        'lr': 0.1,
        'momentum': 0.9,
        'weight_decay': 5e-4,
        'lr_milestones': [50, 75, 90],
        'lr_gamma': 0.1,
    }


def compute_ood_score(model, x):
    """
    Compute OOD score for input batch.

    Returns scores where HIGHER = more likely IN-DISTRIBUTION.

    Args:
        model: trained classifier
        x: input tensor (batch_size, C, H, W)

    Returns:
        scores: tensor of shape (batch_size,)
    """
    was_training = model.training
    model.eval()
    with torch.no_grad():
        if hasattr(model, 'forward_raw'):
            logits = model.forward_raw(x)
        else:
            logits = model(x)
        probs = F.softmax(logits, dim=1)
        scores = probs.max(dim=1).values
    if was_training:
        model.train()
    return scores

"""
OOD detection algorithm: LogitNorm training + MSP scoring (with flip TTA).

LogitNorm (Wei et al., 2022) normalizes the logit vector to a constant norm
(1/tau) during training. Plain cross-entropy keeps increasing the logit norm
to reduce the loss, which yields overconfident predictions even on OOD inputs.
Decoupling the norm from the training objective gives much better separation
between ID and OOD confidence, especially for far-OOD data (e.g. SVHN).

Implementation: the model's forward() returns normalized logits / tau while in
training mode, so the framework's standard cross-entropy becomes the LogitNorm
loss. In eval mode, forward() returns raw logits (argmax is unchanged, so
accuracy is unaffected).

Scoring: MSP on raw logits averaged over the input and its horizontal flip.
Flip averaging reduces per-sample noise in the confidence estimate, which
typically sharpens ID/OOD separation at negligible cost. Set USE_FLIP_TTA=False
to ablate back to plain LogitNorm+MSP.
"""
import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.models as models


LOGITNORM_TAU = 0.04
USE_FLIP_TTA = True


class ResNet18CIFAR(nn.Module):
    """ResNet-18 adapted for CIFAR-10 (32x32 images) with LogitNorm training."""

    def __init__(self, num_classes=10, tau=LOGITNORM_TAU):
        super().__init__()
        backbone = models.resnet18(weights=None)
        # Adapt for CIFAR: smaller conv1, no maxpool
        backbone.conv1 = nn.Conv2d(3, 64, 3, 1, 1, bias=False)
        backbone.maxpool = nn.Identity()
        self.feature_dim = backbone.fc.in_features
        backbone.fc = nn.Identity()
        self.backbone = backbone
        self.fc = nn.Linear(self.feature_dim, num_classes)
        self.tau = tau

    def forward_raw(self, x):
        features = self.backbone(x)
        return self.fc(features)

    def forward(self, x):
        logits = self.forward_raw(x)
        if self.training:
            # LogitNorm: CE on logits normalized to norm 1/tau
            norms = torch.norm(logits, p=2, dim=-1, keepdim=True) + 1e-7
            logits = logits / (norms * self.tau)
        return logits

    def get_features(self, x):
        """Extract penultimate features."""
        return self.backbone(x)


def get_model(num_classes=10):
    """Return classifier model."""
    return ResNet18CIFAR(num_classes=num_classes)


def get_training_config():
    """Return training hyperparameters."""
    return {
        'epochs': 100,
        'batch_size': 128,
        'lr': 0.1,
        'momentum': 0.9,
        'weight_decay': 5e-4,
        'lr_milestones': [50, 75, 90],
        'lr_gamma': 0.1,
    }


def _raw_logits(model, x):
    if hasattr(model, 'forward_raw'):
        return model.forward_raw(x)
    return model(x)


def compute_ood_score(model, x):
    """
    Compute OOD score for input batch.

    Returns scores where HIGHER = more likely IN-DISTRIBUTION.

    Args:
        model: trained classifier
        x: input tensor (batch_size, C, H, W)

    Returns:
        scores: tensor of shape (batch_size,)
    """
    was_training = model.training
    model.eval()
    with torch.no_grad():
        logits = _raw_logits(model, x)
        if USE_FLIP_TTA and x.dim() == 4:
            logits_flip = _raw_logits(model, torch.flip(x, dims=[3]))
            logits = 0.5 * (logits + logits_flip)
        probs = F.softmax(logits, dim=1)
        scores = probs.max(dim=1).values
    if was_training:
        model.train()
    return scores
