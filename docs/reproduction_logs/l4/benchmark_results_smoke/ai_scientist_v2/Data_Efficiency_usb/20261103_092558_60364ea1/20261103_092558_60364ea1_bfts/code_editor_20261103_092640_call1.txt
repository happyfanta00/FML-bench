"""
Improved semi-supervised learning algorithm: FixMatch variant on CIFAR-100.

Changes vs. baseline:
- Lower confidence threshold (0.8) so that, in the very low-label regime
  (4 labels/class), a meaningful amount of pseudo-labels is used early on.
- Larger unlabeled batch (stronger consistency signal per step).
- WRN with LeakyReLU(0.1), low BN momentum and proper init (as in the
  original FixMatch / USB implementations).
- Proper FixMatch-style RandAugment (14 PIL ops, random magnitude) + Cutout.
"""
import random

import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision.transforms as transforms
import numpy as np

try:
    from PIL import Image, ImageOps, ImageEnhance, ImageDraw
    _HAS_PIL = True
except Exception:  # pragma: no cover
    _HAS_PIL = False


BN_MOMENTUM = 0.001
LEAKY = 0.1


class WideResNetBlock(nn.Module):
    """Basic block for WideResNet."""

    def __init__(self, in_channels, out_channels, stride=1, dropout_rate=0.0):
        super().__init__()
        self.bn1 = nn.BatchNorm2d(in_channels, momentum=BN_MOMENTUM)
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3, stride=stride, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels, momentum=BN_MOMENTUM)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3, padding=1, bias=False)
        self.dropout = nn.Dropout(dropout_rate) if dropout_rate > 0 else nn.Identity()

        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Conv2d(in_channels, out_channels, 1, stride=stride, bias=False)

    def forward(self, x):
        out = F.leaky_relu(self.bn1(x), LEAKY)
        shortcut = self.shortcut(out)
        out = self.conv1(out)
        out = self.dropout(F.leaky_relu(self.bn2(out), LEAKY))
        out = self.conv2(out)
        return out + shortcut


class WideResNet(nn.Module):
    """WideResNet-28-2 for CIFAR."""

    def __init__(self, depth=28, widen_factor=2, num_classes=100, dropout_rate=0.0):
        super().__init__()
        assert (depth - 4) % 6 == 0
        n = (depth - 4) // 6
        channels = [16, 16 * widen_factor, 32 * widen_factor, 64 * widen_factor]

        self.conv1 = nn.Conv2d(3, channels[0], 3, padding=1, bias=False)
        self.layer1 = self._make_layer(n, channels[0], channels[1], 1, dropout_rate)
        self.layer2 = self._make_layer(n, channels[1], channels[2], 2, dropout_rate)
        self.layer3 = self._make_layer(n, channels[2], channels[3], 2, dropout_rate)
        self.bn = nn.BatchNorm2d(channels[3], momentum=BN_MOMENTUM)
        self.fc = nn.Linear(channels[3], num_classes)

        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='leaky_relu')
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.constant_(m.weight, 1.0)
                nn.init.constant_(m.bias, 0.0)
            elif isinstance(m, nn.Linear):
                nn.init.xavier_normal_(m.weight)
                nn.init.constant_(m.bias, 0.0)

    def _make_layer(self, n, in_channels, out_channels, stride, dropout_rate):
        layers = [WideResNetBlock(in_channels, out_channels, stride, dropout_rate)]
        for _ in range(1, n):
            layers.append(WideResNetBlock(out_channels, out_channels, 1, dropout_rate))
        return nn.Sequential(*layers)

    def forward(self, x):
        out = self.conv1(x)
        out = self.layer1(out)
        out = self.layer2(out)
        out = self.layer3(out)
        out = F.leaky_relu(self.bn(out), LEAKY)
        out = F.adaptive_avg_pool2d(out, 1).flatten(1)
        return self.fc(out)


def get_model(num_classes=100):
    """Return WideResNet-28-2."""
    return WideResNet(depth=28, widen_factor=2, num_classes=num_classes)


def get_training_config():
    """Return FixMatch training hyperparameters."""
    return {
        'total_steps': 15000,
        'eval_every': 1500,
        'batch_size_labeled': 64,
        'batch_size_unlabeled': 128,
        'lr': 0.03,
        'momentum': 0.9,

        'threshold': 0.8,      # Lower threshold: more pseudo-labels in low-label regime
        'lambda_u': 1.0,       # Weight for unlabeled loss
        'T': 1.0,              # Temperature for sharpening
        'num_labels': 200,     # Total labeled samples (4 per class)
    }


# ----------------------------------------------------------------------------
# FixMatch-style RandAugment on PIL images
# ----------------------------------------------------------------------------
def _rand_sign():
    return 1 if random.random() < 0.5 else -1


def _identity(img, v):
    return img


def _autocontrast(img, v):
    return ImageOps.autocontrast(img)


def _equalize(img, v):
    return ImageOps.equalize(img)


def _brightness(img, v):  # v in [0.05, 0.95]
    return ImageEnhance.Brightness(img).enhance(0.05 + 0.9 * v + 0.0)


def _color(img, v):
    return ImageEnhance.Color(img).enhance(0.05 + 0.9 * v)


def _contrast(img, v):
    return ImageEnhance.Contrast(img).enhance(0.05 + 0.9 * v)


def _sharpness(img, v):
    return ImageEnhance.Sharpness(img).enhance(0.05 + 0.9 * v)


def _posterize(img, v):
    bits = int(4 + round(v * 4))
    bits = max(1, min(8, bits))
    return ImageOps.posterize(img, bits)


def _solarize(img, v):
    return ImageOps.solarize(img, int(256 - v * 256))


def _rotate(img, v):
    return img.rotate(_rand_sign() * v * 30)


def _shear_x(img, v):
    s = _rand_sign() * v * 0.3
    return img.transform(img.size, Image.AFFINE, (1, s, 0, 0, 1, 0))


def _shear_y(img, v):
    s = _rand_sign() * v * 0.3
    return img.transform(img.size, Image.AFFINE, (1, 0, 0, s, 1, 0))


def _translate_x(img, v):
    t = _rand_sign() * v * 0.3 * img.size[0]
    return img.transform(img.size, Image.AFFINE, (1, 0, t, 0, 1, 0))


def _translate_y(img, v):
    t = _rand_sign() * v * 0.3 * img.size[1]
    return img.transform(img.size, Image.AFFINE, (1, 0, 0, 0, 1, t))


def _cutout_pil(img, frac=0.5):
    w, h = img.size
    v = random.uniform(0.0, frac) * min(w, h)
    if v <= 0:
        return img
    x0 = np.random.uniform(0, w)
    y0 = np.random.uniform(0, h)
    x0 = int(max(0, x0 - v / 2.0))
    y0 = int(max(0, y0 - v / 2.0))
    x1 = int(min(w, x0 + v))
    y1 = int(min(h, y0 + v))
    img = img.copy()
    fill = (127, 127, 127) if img.mode == 'RGB' else 127
    ImageDraw.Draw(img).rectangle((x0, y0, x1, y1), fill=fill)
    return img


_PIL_OPS = [
    _identity, _autocontrast, _equalize, _brightness, _color, _contrast,
    _sharpness, _posterize, _solarize, _rotate, _shear_x, _shear_y,
    _translate_x, _translate_y,
]


class RandAugment:
    """FixMatch-style RandAugment (random magnitude) + Cutout."""

    def __init__(self, n=2, m=10):
        self.n = n
        self.m = m
        # Fallback tensor-compatible ops (used if input isn't a PIL image)
        self.augmentations = [
            transforms.RandomAffine(degrees=15, translate=(0.1, 0.1)),
            transforms.RandomHorizontalFlip(p=1.0),
            transforms.ColorJitter(brightness=0.4, contrast=0.4, saturation=0.4),
            transforms.RandomGrayscale(p=1.0),
            transforms.RandomAutocontrast(p=1.0),
        ]
        self.erase = transforms.RandomErasing(p=1.0, scale=(0.02, 0.25), value=0.5)

    def __call__(self, img):
        if _HAS_PIL and isinstance(img, Image.Image):
            try:
                if img.mode != 'RGB':
                    img = img.convert('RGB')
                ops = random.sample(_PIL_OPS, self.n)
                for op in ops:
                    v = random.uniform(0.05, 0.95)
                    img = op(img, v)
                img = _cutout_pil(img, 0.5)
                return img
            except Exception:
                return img
        # Tensor fallback
        ops = np.random.choice(len(self.augmentations), self.n, replace=False)
        for i in ops:
            try:
                img = self.augmentations[i](img)
            except Exception:
                pass
        if torch.is_tensor(img) and img.is_floating_point():
            try:
                img = self.erase(img)
            except Exception:
                pass
        return img
