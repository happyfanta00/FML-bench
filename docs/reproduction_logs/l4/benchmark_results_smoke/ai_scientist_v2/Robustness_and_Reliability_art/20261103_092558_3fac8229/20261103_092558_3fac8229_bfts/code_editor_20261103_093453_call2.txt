This version adds a poison-filtering stage before DP-InstaHide training: a short clean warm-up, followed by
per-class feature-space outlier detection (spectral signatures + activation clustering) and label-consistency
filtering, a reset of the model weights and finally DP-InstaHide training on the filtered data.

class DPInstaHideTrainer(Trainer):
    """
    Class performing adversarial training following the DP-InstaHide protocol.

    Uses data augmentation methods in conjunction with some type of additive noise.

    | Paper link: https://arxiv.org/abs/2103.02079
    """

    def __init__(
        self,
        classifier: "CLASSIFIER_LOSS_GRADIENTS_TYPE",
        augmentations: Union["Preprocessor", List["Preprocessor"]],
        noise: Literal["gaussian", "laplacian", "exponential"] = "laplacian",
        loc: Union[int, float] = 0.0,
        scale: Union[int, float] = 0.03,
        clip_values: "CLIP_VALUES_TYPE" = (0.0, 1.0),
    ):
        """
        Create an :class:`.DPInstaHideTrainer` instance.

        :param classifier: The model to train using the protocol.
        :param augmentations: The preprocessing data augmentation defence(s) to be applied.
        :param noise: The type of additive noise to use: 'gaussian' | 'laplacian' | 'exponential'.
        :param loc: The location or mean parameter of the distribution to sample.
        :param scale: The scale or standard deviation parameter of the distribution to sample.
        :param clip_values: Tuple of the form `(min, max)` representing the minimum and maximum values allowed
               for features.
        """
        from art.defences.preprocessor import Preprocessor

        super().__init__(classifier)
        if isinstance(augmentations, Preprocessor):
            self.augmentations = [augmentations]
        else:
            self.augmentations = augmentations
        self.noise = noise
        self.loc = loc
        self.scale = scale
        self.clip_values = clip_values

        # filtering hyper-parameters
        self.warmup_epochs = 2
        self.spectral_z = 3.5
        self.max_spectral_frac = 0.15
        self.ac_size_ratio = 0.25
        self.ac_min_silhouette = 0.10
        self.label_prob_thresh = 0.01
        self.max_label_frac = 0.05

    def _generate_noise(self, x: np.ndarray) -> np.ndarray:
        if self.noise == "gaussian":
            noise = np.random.normal(loc=self.loc, scale=self.scale, size=x.shape)
        elif self.noise == "laplacian":
            noise = np.random.laplace(loc=self.loc, scale=self.scale, size=x.shape)
        elif self.noise == "exponential":
            noise = np.random.exponential(scale=self.scale, size=x.shape)
        else:
            raise ValueError("The provided noise type is not supported:", self.noise)

        x_noise = x + noise
        x_noise = np.clip(x_noise, self.clip_values[0], self.clip_values[1])

        return x_noise

    # ------------------------------------------------------------------ #
    # Poison filtering helpers
    # ------------------------------------------------------------------ #
    def _get_weights(self):
        try:
            model = getattr(self._classifier, "model", None)
            if model is None:
                model = getattr(self._classifier, "_model", None)
            if model is not None and hasattr(model, "get_weights"):
                return model, [w.copy() for w in model.get_weights()]
        except Exception:  # pylint: disable=W0703
            pass
        return None, None

    def _get_features(self, x: np.ndarray, batch_size: int) -> np.ndarray:
        feats = None
        try:
            layer_names = self._classifier.layer_names
            if layer_names is not None and len(layer_names) >= 3:
                # penultimate layer (dropout of dense-128 -> identity at inference)
                idx = len(layer_names) - 2
                feats = self._classifier.get_activations(x, idx, batch_size=batch_size)
                feats = np.asarray(feats).reshape(len(x), -1)
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Could not obtain activations, falling back to outputs: %s", err)
            feats = None
        if feats is None or feats.shape[0] != len(x) or not np.all(np.isfinite(feats)):
            probs = self._classifier.predict(x, batch_size=batch_size)
            feats = np.log(np.clip(probs, 1e-12, 1.0)).reshape(len(x), -1)
        return feats.astype(np.float64)

    def _filter_class(self, feats: np.ndarray) -> np.ndarray:
        """Return boolean mask of suspicious samples within one class."""
        n = feats.shape[0]
        suspicious = np.zeros(n, dtype=bool)
        if n < 20:
            return suspicious
        centered = feats - feats.mean(axis=0, keepdims=True)
        try:
            _, _, vt = np.linalg.svd(centered, full_matrices=False)
        except np.linalg.LinAlgError:
            return suspicious

        # Spectral signature
        proj = np.abs(centered @ vt[0])
        med = np.median(proj)
        mad = np.median(np.abs(proj - med)) + 1e-12
        z = (proj - med) / (1.4826 * mad)
        cand = np.where(z > self.spectral_z)[0]
        max_rm = int(self.max_spectral_frac * n)
        if len(cand) > max_rm:
            cand = cand[np.argsort(-z[cand])[:max_rm]]
        suspicious[cand] = True

        # Activation clustering
        try:
            from sklearn.cluster import KMeans
            from sklearn.metrics import silhouette_score

            n_comp = min(10, vt.shape[0])
            red = centered @ vt[:n_comp].T
            km = KMeans(n_clusters=2, n_init=5, random_state=0).fit(red)
            labels = km.labels_
            sizes = np.bincount(labels, minlength=2)
            small = int(np.argmin(sizes))
            ratio = sizes[small] / float(n)
            if 0 < ratio < self.ac_size_ratio:
                rng = np.random.RandomState(0)
                sub = rng.choice(n, size=min(n, 2000), replace=False)
                if len(np.unique(labels[sub])) == 2:
                    sil = silhouette_score(red[sub], labels[sub])
                    if sil > self.ac_min_silhouette:
                        suspicious[labels == small] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Activation clustering failed: %s", err)

        return suspicious

    def _filter_poisons(self, x: np.ndarray, y: np.ndarray, batch_size: int) -> np.ndarray:
        """Return indices of samples to keep."""
        n = len(x)
        labels = np.argmax(y, axis=1) if y.ndim > 1 else y.astype(int)
        remove = np.zeros(n, dtype=bool)

        # label consistency
        try:
            probs = self._classifier.predict(x, batch_size=batch_size)
            p_true = probs[np.arange(n), labels]
            cand = np.where(p_true < self.label_prob_thresh)[0]
            max_rm = int(self.max_label_frac * n)
            if len(cand) > max_rm:
                cand = cand[np.argsort(p_true[cand])[:max_rm]]
            remove[cand] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Label-consistency filtering failed: %s", err)

        # feature-space filtering per class
        try:
            feats = self._get_features(x, batch_size)
            for c in np.unique(labels):
                idx = np.where(labels == c)[0]
                sus = self._filter_class(feats[idx])
                remove[idx[sus]] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Feature filtering failed: %s", err)

        # never remove too much
        if remove.sum() > 0.4 * n:
            logger.warning("Too many removals (%d), skipping filtering.", int(remove.sum()))
            return np.arange(n)
        logger.info("Poison filtering removed %d / %d samples", int(remove.sum()), n)
        return np.where(~remove)[0]

    def fit(  # pylint: disable=W0221
        self,
        x: np.ndarray,
        y: np.ndarray,
        validation_data: Optional[Tuple[np.ndarray, np.ndarray]] = None,
        batch_size: int = 128,
        nb_epochs: int = 20,
        **kwargs
    ):
        """
        Train a model adversarially with the DP-InstaHide protocol.
        See class documentation for more information on the exact procedure.

        :param x: Training set.
        :param y: Labels for the training set.
        :param validation_data: Tuple consisting of validation data, (x_val, y_val)
        :param batch_size: Size of batches.
        :param nb_epochs: Number of epochs to use for trainings.
        :param kwargs: Dictionary of framework-specific arguments. These will be passed as such to the `fit` function
               of the target classifier.
        """
        logger.info("Performing adversarial training with DP-InstaHide protocol")

        # ---------------- Stage 1: warm-up + poison filtering ----------------
        model, init_weights = self._get_weights()
        try:
            self._classifier.fit(
                x, y, nb_epochs=self.warmup_epochs, batch_size=batch_size, verbose=0, **kwargs
            )
            keep = self._filter_poisons(x, y, batch_size)
            x = x[keep]
            y = y[keep]
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Warm-up / filtering stage failed: %s", err)

        # reset weights to initialisation so that the warm-up (possibly backdoored) model is discarded
        if model is not None and init_weights is not None:
            try:
                model.set_weights(init_weights)
            except Exception as err:  # pylint: disable=W0703
                logger.warning("Could not reset weights: %s", err)

        # ---------------- Stage 2: DP-InstaHide training ----------------
        nb_batches = int(np.ceil(len(x) / batch_size))
        ind = np.arange(len(x))

        logger.info("Adversarial Training DP-InstaHide")

        for i_epoch in trange(nb_epochs, desc="DP-InstaHide training epochs"):
            # shuffle the examples
            np.random.shuffle(ind)
            start_time = time.time()
            train_loss = 0.0
            train_acc = 0.0
            train_n = 0.0

            for batch_id in range(nb_batches):
                # create batch data
                x_batch = x[ind[batch_id * batch_size : min((batch_id + 1) * batch_size, x.shape[0])]]
                y_batch = y[ind[batch_id * batch_size : min((batch_id + 1) * batch_size, x.shape[0])]]

                # augment the data
                x_aug = x_batch.copy()
                y_aug = y_batch.copy()
                for augmentation in self.augmentations:
                    x_aug, y_aug = augmentation(x_aug, y_aug)

                # apply additive noise
                x_aug = self._generate_noise(x_aug)

                # fit batch
                self._classifier.fit(x_aug, y_aug, nb_epochs=1, batch_size=x_aug.shape[0], verbose=0, **kwargs)

                # get metrics
                loss = self._classifier.compute_loss(x_aug, y_aug, reduction="mean")
                output = np.argmax(self.predict(x_batch), axis=1)
                acc = np.sum(output == np.argmax(y_batch, axis=1))
                n = len(x_aug)

                # update metrics
                train_loss += np.sum(loss)
                train_acc += acc
                train_n += n

            train_time = time.time()

            # compute accuracy
            if validation_data is not None:
                (x_test, y_test) = validation_data
                output = np.argmax(self.predict(x_test), axis=1)
                test_loss = self._classifier.compute_loss(x_test, y_test, reduction="mean")
                test_acc = np.mean(output == np.argmax(y_test, axis=1))
                logger.info(
                    "epoch: %s time(s): %.1f, loss(tr): %.4f, acc(tr): %.4f, loss(val): %.4f, acc(val): %.4f",
                    i_epoch,
                    train_time - start_time,
                    train_loss / train_n,
                    train_acc / train_n,
                    test_loss,
                    test_acc,
                )
            else:
                logger.info(
                    "epoch: %s time(s): %.1f, loss: %.4f, acc: %.4f",
                    i_epoch,
                    train_time - start_time,
                    train_loss / train_n,
                    train_acc / train_n,
                )

    def fit_generator(self, generator: "DataGenerator", nb_epochs: int = 20, **kwargs):
        """

        self.ac_small_ratio = 0.25
        self.ac_min_silhouette = 0.10
        self.ac_min_silhouette_large = 0.20
        self.label_prob_thresh = 0.01
        self.max_label_frac = 0.05
        self.max_total_frac = 0.5

    def _generate_noise(self, x: np.ndarray) -> np.ndarray:
        if self.noise == "gaussian":
            noise = np.random.normal(loc=self.loc, scale=self.scale, size=x.shape)
        elif self.noise == "laplacian":
            noise = np.random.laplace(loc=self.loc, scale=self.scale, size=x.shape)
        elif self.noise == "exponential":
            noise = np.random.exponential(scale=self.scale, size=x.shape)
        else:
            raise ValueError("The provided noise type is not supported:", self.noise)

        x_noise = x + noise
        x_noise = np.clip(x_noise, self.clip_values[0], self.clip_values[1])

        return x_noise

    # ------------------------------------------------------------------ #
    # Poison filtering helpers
    # ------------------------------------------------------------------ #
    def _get_weights(self):
        try:
            model = getattr(self._classifier, "model", None)
            if model is None:
                model = getattr(self._classifier, "_model", None)
            if model is not None and hasattr(model, "get_weights"):
                return model, [w.copy() for w in model.get_weights()]
        except Exception:  # pylint: disable=W0703
            pass
        return None, None

    def _get_features(self, x: np.ndarray, batch_size: int) -> np.ndarray:
        feats = None
        try:
            layer_names = self._classifier.layer_names
            if layer_names is not None and len(layer_names) >= 3:
                # penultimate layer (dropout of dense-128 -> identity at inference)
                idx = len(layer_names) - 2
                feats = self._classifier.get_activations(x, idx, batch_size=batch_size)
                feats = np.asarray(feats).reshape(len(x), -1)
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Could not obtain activations, falling back to outputs: %s", err)
            feats = None
        if feats is None or feats.shape[0] != len(x) or not np.all(np.isfinite(feats)):
            probs = self._classifier.predict(x, batch_size=batch_size)
            feats = np.log(np.clip(probs, 1e-12, 1.0)).reshape(len(x), -1)
        return feats.astype(np.float64)

    def _filter_class(self, feats: np.ndarray) -> np.ndarray:
        """Return boolean mask of suspicious samples within one class."""
        n = feats.shape[0]
        suspicious = np.zeros(n, dtype=bool)
        if n < 20:
            return suspicious
        centered = feats - feats.mean(axis=0, keepdims=True)
        try:
            _, _, vt = np.linalg.svd(centered, full_matrices=False)
        except np.linalg.LinAlgError:
            return suspicious

        # Spectral signature (robust z-score of projection on top singular vector)
        proj = np.abs(centered @ vt[0])
        med = np.median(proj)
        mad = np.median(np.abs(proj - med)) + 1e-12
        z = (proj - med) / (1.4826 * mad)
        cand = np.where(z > self.spectral_z)[0]
        max_rm = int(self.max_spectral_frac * n)
        if len(cand) > max_rm:
            cand = cand[np.argsort(-z[cand])[:max_rm]]
        suspicious[cand] = True

        # Activation clustering
        try:
            from sklearn.cluster import KMeans
            from sklearn.metrics import silhouette_score

            n_comp = min(10, vt.shape[0])
            red = centered @ vt[:n_comp].T
            km = KMeans(n_clusters=2, n_init=5, random_state=0).fit(red)
            labels = km.labels_
            sizes = np.bincount(labels, minlength=2)
            small = int(np.argmin(sizes))
            ratio = sizes[small] / float(n)
            if 0 < ratio < self.ac_size_ratio:
                # larger "small" clusters need stronger separation to avoid removing clean sub-modes
                sil_thresh = self.ac_min_silhouette if ratio < self.ac_small_ratio else self.ac_min_silhouette_large
                rng = np.random.RandomState(0)
                sub = rng.choice(n, size=min(n, 2000), replace=False)
                if len(np.unique(labels[sub])) == 2:
                    sil = silhouette_score(red[sub], labels[sub])
                    if sil > sil_thresh:
                        suspicious[labels == small] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Activation clustering failed: %s", err)

        return suspicious

    def _filter_poisons(self, x: np.ndarray, y: np.ndarray, batch_size: int) -> np.ndarray:
        """Return indices of samples to keep."""
        n = len(x)
        labels = np.argmax(y, axis=1) if y.ndim > 1 else y.astype(int)
        remove = np.zeros(n, dtype=bool)

        # label consistency
        try:
            probs = self._classifier.predict(x, batch_size=batch_size)
            p_true = probs[np.arange(n), labels]
            cand = np.where(p_true < self.label_prob_thresh)[0]
            max_rm = int(self.max_label_frac * n)
            if len(cand) > max_rm:
                cand = cand[np.argsort(p_true[cand])[:max_rm]]
            remove[cand] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Label-consistency filtering failed: %s", err)

        # feature-space filtering per class
        try:
            feats = self._get_features(x, batch_size)
            for c in np.unique(labels):
                idx = np.where(labels == c)[0]
                sus = self._filter_class(feats[idx])
                remove[idx[sus]] = True
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Feature filtering failed: %s", err)

        # never remove too much
        if remove.sum() > self.max_total_frac * n:
            logger.warning("Too many removals (%d), skipping filtering.", int(remove.sum()))
            return np.arange(n)
        logger.info("Poison filtering removed %d / %d samples", int(remove.sum()), n)
        return np.where(~remove)[0]

    def fit(  # pylint: disable=W0221
        self,
        x: np.ndarray,
        y: np.ndarray,
        validation_data: Optional[Tuple[np.ndarray, np.ndarray]] = None,
        batch_size: int = 128,
        nb_epochs: int = 20,
        **kwargs
    ):
        """
        Train a model adversarially with the DP-InstaHide protocol.
        See class documentation for more information on the exact procedure.

        :param x: Training set.
        :param y: Labels for the training set.
        :param validation_data: Tuple consisting of validation data, (x_val, y_val)
        :param batch_size: Size of batches.
        :param nb_epochs: Number of epochs to use for trainings.
        :param kwargs: Dictionary of framework-specific arguments. These will be passed as such to the `fit` function
               of the target classifier.
        """
        logger.info("Performing adversarial training with DP-InstaHide protocol")
        kwargs = dict(kwargs)
        kwargs.pop("verbose", None)

        # ---------------- Stage 1: warm-up + poison filtering ----------------
        model, init_weights = self._get_weights()
        try:
            self._classifier.fit(
                x, y, nb_epochs=self.warmup_epochs, batch_size=batch_size, verbose=0, **kwargs
            )
            keep = self._filter_poisons(x, y, batch_size)
            x = x[keep]
            y = y[keep]
        except Exception as err:  # pylint: disable=W0703
            logger.warning("Warm-up / filtering stage failed: %s", err)

        # reset weights to initialisation so that the warm-up (possibly backdoored) model is discarded
        if model is not None and init_weights is not None:
            try:
                model.set_weights(init_weights)
            except Exception as err:  # pylint: disable=W0703
                logger.warning("Could not reset weights: %s", err)

        # ---------------- Stage 2: DP-InstaHide training ----------------
        nb_batches = int(np.ceil(len(x) / batch_size))
        ind = np.arange(len(x))

        logger.info("Adversarial Training DP-InstaHide")

        for i_epoch in trange(nb_epochs, desc="DP-InstaHide training epochs"):
            # shuffle the examples
            np.random.shuffle(ind)
            start_time = time.time()
            train_loss = 0.0
            train_n = 0.0

            for batch_id in range(nb_batches):
                # create batch data
                x_batch = x[ind[batch_id * batch_size : min((batch_id + 1) * batch_size, x.shape[0])]]
                y_batch = y[ind[batch_id * batch_size : min((batch_id + 1) * batch_size, x.shape[0])]]

                # augment the data
                x_aug = x_batch.copy()
                y_aug = y_batch.copy()
                for augmentation in self.augmentations:
                    x_aug, y_aug = augmentation(x_aug, y_aug)

                # apply additive noise
                x_aug = self._generate_noise(x_aug)

                # fit batch
                self._classifier.fit(x_aug, y_aug, nb_epochs=1, batch_size=x_aug.shape[0], verbose=0, **kwargs)

                # get metrics
                loss = self._classifier.compute_loss(x_aug, y_aug, reduction="mean")
                train_loss += np.sum(loss) * len(x_aug)
                train_n += len(x_aug)

            train_time = time.time()

            # compute accuracy
            if validation_data is not None:
                (x_test, y_test) = validation_data
                output = np.argmax(self.predict(x_test), axis=1)
                test_loss = self._classifier.compute_loss(x_test, y_test, reduction="mean")
                test_acc = np.mean(output == np.argmax(y_test, axis=1))
                logger.info(
                    "epoch: %s time(s): %.1f, loss(tr): %.4f, loss(val): %.4f, acc(val): %.4f",
                    i_epoch,
                    train_time - start_time,
                    train_loss / max(train_n, 1.0),
                    test_loss,
                    test_acc,
                )
            else:
                logger.info(
                    "epoch: %s time(s): %.1f, loss: %.4f",
                    i_epoch,
                    train_time - start_time,
                    train_loss / max(train_n, 1.0),
                )

    def fit_generator(self, generator: "DataGenerator", nb_epochs: int = 20, **kwargs):
        """
        Train a model adversarially with the DP-InstaHide protocol using a data generator.
        See class documentation for more information on the exact procedure.

        :param generator: Data generator.
        :param nb_epochs: Number of epochs to use for trainings.
        :param kwargs: Dictionary of framework-specific arguments. These will be passed as such to the `fit` function
               of the target classifier.
        """
        logger.info("Performing adversarial training with DP-InstaHide protocol")
        kwargs = dict(kwargs)
        kwargs.pop("verbose", None)
        size = generator.size
        batch_size = generator.batch_size
        if size is not None:
            nb_batches = int(np.ceil(size / batch_size))
        else:
            raise ValueError("Size is None.")

        logger.info("Adversarial Training DP-InstaHide")

        for i_epoch in trange(nb_epochs, desc="DP-InstaHide training epochs"):
            start_time = time.time()
            train_loss = 0.0
            train_acc = 0.0
            train_n = 0.0

            for _ in range(nb_batches):
                # create batch data
                x_batch, y_batch = generator.get_batch()

                # augment the data
                x_aug = x_batch.copy()
                y_aug = y_batch.copy()
                for augmentation in self.augmentations:
                    x_aug, y_aug = augmentation(x_aug, y_aug)

                # apply additive noise
                x_aug = self._generate_noise(x_aug)

                # fit batch
                self._classifier.fit(x_aug, y_aug, nb_epochs=1, batch_size=x_aug.shape[0], verbose=0, **kwargs)

                # get metrics
                loss = self._classifier.compute_loss(x_aug, y_aug, reduction="mean")
                output = np.argmax(self.predict(x_batch), axis=1)
                acc = np.sum(output == np.argmax(y_batch, axis=1))
                n = len(x_aug)

                # update metrics
                train_loss += np.sum(loss)
                train_acc += acc
                train_n += n

            train_time = time.time()

            # compute accuracy
            logger.info(
                "epoch: %s time(s): %.1f, loss: %.4f, acc: %.4f",
                i_epoch,
                train_time - start_time,
                train_loss / train_n,
                train_acc / train_n,
            )

    def predict(self, x: np.ndarray, **kwargs) -> np.ndarray:
        """
        Perform prediction using the adversarially trained classifier.

        :param x: Input samples.
        :param kwargs: Other parameters to be passed on to the `predict` function of the classifier.
        :return: Predictions for test set.
        """
        return self._classifier.predict(x, **kwargs)
