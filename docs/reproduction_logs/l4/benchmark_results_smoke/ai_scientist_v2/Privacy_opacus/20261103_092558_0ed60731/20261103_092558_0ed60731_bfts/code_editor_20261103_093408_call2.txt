"""
DP training algorithm: Tramer & Boneh-style CNN with GroupNorm + tempered (tanh)
activations, trained with Opacus DP-SGD on CIFAR-10 at epsilon=8.

Changes vs. baseline:
- Deeper VGG-style CNN (6 conv layers, 32-64-128 channels) as in
  Tramer & Boneh (2021), "Differentially Private Learning Needs Better Features".
- Tanh activations ("tempered sigmoids", Papernot et al. 2021), which keep
  activations and per-sample gradient norms bounded. This makes clipping
  less destructive.
- GroupNorm after every conv (DP-compatible normalization).
- Larger batch size (better signal-to-noise ratio per step) and more epochs.
"""
import torch
import torch.nn as nn


def _conv_block(in_c, out_c, groups=8):
    return [
        nn.Conv2d(in_c, out_c, 3, padding=1),
        nn.GroupNorm(groups, out_c),
        nn.Tanh(),
    ]


class DPCNN(nn.Module):
    """
    VGG-style CNN with GroupNorm and Tanh activations for DP-SGD.
    """

    def __init__(self, num_classes=10):
        super().__init__()
        layers = []
        layers += _conv_block(3, 32)
        layers += _conv_block(32, 32)
        layers += [nn.MaxPool2d(2)]          # 16x16
        layers += _conv_block(32, 64)
        layers += _conv_block(64, 64)
        layers += [nn.MaxPool2d(2)]          # 8x8
        layers += _conv_block(64, 128)
        layers += _conv_block(128, 128)
        layers += [nn.AdaptiveAvgPool2d((4, 4))]  # 4x4
        self.features = nn.Sequential(*layers)
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 4 * 4, 128),
            nn.Tanh(),
            nn.Linear(128, num_classes),
        )

    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x


# Keep the old name available for compatibility.
SimpleCNN = DPCNN


def get_model(num_classes=10):
    """Return the model to be trained with DP-SGD."""
    return DPCNN(num_classes=num_classes)

        'momentum': 0.9,
        'max_grad_norm': 1.0,
        'target_epsilon': 8.0,
        'target_delta': 1e-5,
        'batch_size': 1024,
    }
